\begin{notation}

  \begin{notationlist}{2em}
    \item[$\displaystyle \mathcal{U}, \mathcal{I}$] 用户集合、物品集合
    \item[$\displaystyle \mathcal{O}$] 用户—物品交互集合
    \item[$\displaystyle \mathcal{H}_u$] 用户 $u$ 的历史交互集合
    \item[$\displaystyle \mathbf{X}$] 用户—物品交互矩阵
    \item[$\displaystyle \mathbf{e}_u, \mathbf{e}_i$] 用户、物品的协同过滤嵌入
    \item[$\displaystyle \mathbf{p}_i$] 物品 $i$ 的静态语义画像向量
    \item[$\displaystyle \mathbf{d}_u, \mathbf{d}_i$] 用户、物品的动态画像向量
    \item[$\displaystyle \mathbf{c}_k$] 第 $k$ 个兴趣原型中心
    \item[$\displaystyle \mathbf{h}_u$] 用户 $u$ 的兴趣表征
    \item[$\displaystyle \alpha_{j}$] 注意力权重
    \item[$\displaystyle \tau$] 对比学习温度系数
    \item[$\displaystyle \mathcal{L}_{\mathrm{contra}}$] 语义难负例对比损失
    \item[$\displaystyle \mathcal{N}_u^{\ast}$] 用户 $u$ 在某个视角下的邻域集合
    \item[$\displaystyle w_m$] 第 $m$ 个视角的表征融合权重
    \item[$\displaystyle \hat{y}_{ui}$] 用户 $u$ 对物品 $i$ 的点击概率预测值
  \end{notationlist}

\end{notation}
